# Интеграция с 1С

## Полный рабочий код HTTP-сервиса

### Шаг 1: Создание HTTP-сервиса в конфигураторе

1. Откройте конфигуратор 1С
2. Добавьте новый HTTP-сервис:
   - **Имя:** `WebAPI`
   - **Корневой URL:** `api`

### Шаг 2: Создание URL-шаблонов

Добавьте следующие URL-шаблоны в HTTP-сервис:

| Шаблон | Метод | Имя обработчика |
|--------|-------|-----------------|
| `/ping` | GET | `Ping` |
| `/categories` | GET | `ПолучитьКатегории` |
| `/products` | GET | `ПолучитьТовары` |
| `/stock` | GET | `ПолучитьОстатки` |

### Шаг 3: Полный код модуля HTTP-сервиса

Скопируйте ВЕСЬ код ниже в модуль HTTP-сервиса:

```1c
//===============================================================================
// МОДУЛЬ HTTP-СЕРВИСА WebAPI
// Интеграция с интернет-магазином KMO24
// Версия: 2.0
//===============================================================================

// =============================================================================
// ОБРАБОТЧИК: Ping
// URL: /ping, Метод: GET

Функция Ping(Запрос) Экспорт

    Ответ = Новый HTTPСервисОтвет(200);
    Ответ.Заголовки.Вставить("Content-Type", "application/json; charset=utf-8");

    Данные = Новый Структура;
    Данные.Вставить("success", Истина);
    Данные.Вставить("status", "ok");
    Данные.Вставить("timestamp", XMLСтрока(ТекущаяДатаСеанса()));

    ЗаписьJSON = Новый ЗаписьJSON;
    ЗаписьJSON.УстановитьСтроку();
    ЗаписатьJSON(ЗаписьJSON, Данные);

    Ответ.УстановитьТелоИзСтроки(ЗаписьJSON.Закрыть(), "UTF-8");

    Возврат Ответ;

// =============================================================================

        КонецЕсли;

        "ВЫБРАТЬ
        |    Номенклатура.Ссылка КАК Ссылка,
        |    Номенклатура.Код КАК Код,
        |    Номенклатура.Наименование КАК Наименование,
        |    Номенклатура.Артикул КАК Артикул,
        |    Номенклатура.НаименованиеПолное КАК НаименованиеПолное,
        |    Номенклатура.Комментарий КАК Описание,
        |    Номенклатура.КатегорияНоменклатуры КАК Категория,
        |    Номенклатура.КатегорияНоменклатуры.Наименование КАК КатегорияНаименование,
        |    Номенклатура.ЕдиницаИзмерения.Наименование КАК ЕдиницаИзмерения,
        |    Номенклатура.ПометкаУдаления КАК ПометкаУдаления
        |ИЗ
        |    Справочник.Номенклатура КАК Номенклатура
        |ГДЕ
        |    Номенклатура.ЭтоГруппа = ЛОЖЬ
        |    И НЕ Номенклатура.ПометкаУдаления";

        // Фильтр по категории
        Если ЗначениеЗаполнено(КатегорияID) Тогда
            Попытка
                КатегорияСсылка = Справочники.КатегорииНоменклатуры.ПолучитьСсылку(
                    Новый УникальныйИдентификатор(КатегорияID));
                Если ЗначениеЗаполнено(КатегорияСсылка) Тогда
                    ТекстЗапроса = ТекстЗапроса + "
                    |    И Номенклатура.КатегорияНоменклатуры = &Категория";
                    Запрос1С.УстановитьПараметр("Категория", КатегорияСсылка);
                КонецЕсли;
            Исключение
            КонецПопытки;
        КонецЕсли;

        ТекстЗапроса = ТекстЗапроса + "
        |УПОРЯДОЧИТЬ ПО
        |    Номенклатура.Наименование";

        Запрос1С.Текст = ТекстЗапроса;
        Результат = Запрос1С.Выполнить();
        Выборка = Результат.Выбрать();

        // Пагинация
        Смещение = (Страница - 1) * Лимит;
        Счетчик = 0;
        ВсегоЗаписей = 0;

        МассивТоваров = Новый Массив;

        Пока Выборка.Следующий() Цикл
            ВсегоЗаписей = ВсегоЗаписей + 1;

            Если ВсегоЗаписей <= Смещение Тогда
                Продолжить;
            КонецЕсли;

                Продолжить; // Продолжаем считать общее количество
            КонецЕсли;

            // Получаем цену
            Цена = ПолучитьЦенуНоменклатуры(Выборка.Ссылка);

            // Получаем остаток
            Остаток = ПолучитьОстатокНоменклатуры(Выборка.Ссылка);

            Товар = Новый Структура;
            Товар.Вставить("id", Строка(Выборка.Ссылка.УникальныйИдентификатор()));
            Товар.Вставить("sku", СокрЛП(Выборка.Артикул));
            Товар.Вставить("code", СокрЛП(Выборка.Код));
            Товар.Вставить("name", СокрЛП(Выборка.Наименование));
            Товар.Вставить("fullName", СокрЛП(Выборка.НаименованиеПолное));
            Товар.Вставить("description", СокрЛП(Выборка.Описание));
            Товар.Вставить("price", Цена);
            Товар.Вставить("stock", Остаток);
            Товар.Вставить("unit", СокрЛП(Выборка.ЕдиницаИзмерения));
            Товар.Вставить("isActive", НЕ Выборка.ПометкаУдаления);

            Если ЗначениеЗаполнено(Выборка.Категория) Тогда
                Товар.Вставить("categoryId", Строка(Выборка.Категория.УникальныйИдентификатор()));
                Товар.Вставить("categoryName", СокрЛП(Выборка.КатегорияНаименование));
            Иначе
                Товар.Вставить("categoryId", "");
                Товар.Вставить("categoryName", "");
            КонецЕсли;

            // Получаем изображения товара
            МассивИзображений = ПолучитьИзображенияНоменклатуры(Выборка.Ссылка);
            Товар.Вставить("images", МассивИзображений);

            МассивТоваров.Добавить(Товар);
        КонецЦикла;

        ОтветДанные = Новый Структура;
        ОтветДанные.Вставить("success", Истина);
        ОтветДанные.Вставить("data", МассивТоваров);
        ОтветДанные.Вставить("pagination", Новый Структура("page,limit,total",
            Страница, Лимит, ВсегоЗаписей));

        Возврат СоздатьОтвет(200, ОтветДанные);

    Исключение
        Возврат СоздатьОтвет(500, Неопределено, ОписаниеОшибки());
    КонецПопытки;

Функция СоздатьОтвет(КодСостояния, Данные, Ошибка = "")

    Ответ = Новый HTTPСервисОтвет(КодСостояния);
    Ответ.Заголовки.Вставить("Content-Type", "application/json; charset=utf-8");

    ТелоОтвета = Новый Структура;

    Если КодСостояния >= 200 И КодСостояния < 300 Тогда
        Если Данные <> Неопределено Тогда
            Если ТипЗнч(Данные) = Тип("Структура") И Данные.Свойство("success") Тогда
                ТелоОтвета = Данные;
            Иначе
                ТелоОтвета.Вставить("success", Истина);
                ТелоОтвета.Вставить("data", Данные);
            КонецЕсли;
        Иначе
            ТелоОтвета.Вставить("success", Истина);
        КонецЕсли;
    Иначе
        ТелоОтвета.Вставить("success", Ложь);
        ТелоОтвета.Вставить("error", Ошибка);
    КонецЕсли;

    ЗаписьJSON = Новый ЗаписьJSON;
    ЗаписьJSON.УстановитьСтроку();
    ЗаписатьJSON(ЗаписьJSON, ТелоОтвета);

    Ответ.УстановитьТелоИзСтроки(ЗаписьJSON.Закрыть(), "UTF-8");

    Возврат Ответ;

// =============================================================================
// ПОЛУЧЕНИЕ ИЗОБРАЖЕНИЙ НОМЕНКЛАТУРЫ
// Возвращает массив URL или Base64 изображений товара
// =============================================================================
Функция ПолучитьИзображенияНоменклатуры(НоменклатураСсылка)

    МассивИзображений = Новый Массив;

    Попытка
        // Вариант 1: Если изображения хранятся в регистре сведений "ФайлыНоменклатуры"
        // (типичная схема для 1С:Управление торговлей, 1С:ERP)
        Запрос = Новый Запрос;
        Запрос.Текст =
        "ВЫБРАТЬ
        |    ФайлыНоменклатуры.Файл КАК Файл,
        |    ФайлыНоменклатуры.Файл.Расширение КАК Расширение
        |ИЗ
        |    РегистрСведений.ФайлыНоменклатуры КАК ФайлыНоменклатуры
        |ГДЕ
        |    ФайлыНоменклатуры.Номенклатура = &Номенклатура";

        Запрос.УстановитьПараметр("Номенклатура", НоменклатураСсылка);

        РезультатЗапроса = Запрос.Выполнить();

        Если НЕ РезультатЗапроса.Пустой() Тогда
            Выборка = РезультатЗапроса.Выбрать();

            Пока Выборка.Следующий() Цикл
                Если ЗначениеЗаполнено(Выборка.Файл) Тогда
                    // Получаем двоичные данные файла
                    ДанныеФайла = ПолучитьДвоичныеДанныеФайла(Выборка.Файл);

                    Если ДанныеФайла <> Неопределено Тогда
                        // Кодируем в Base64
                        СтрокаBase64 = Base64Строка(ДанныеФайла);

                        // Определяем MIME-тип
                        Расширение = НРег(СокрЛП(Выборка.Расширение));
                        Если Расширение = "jpg" ИЛИ Расширение = "jpeg" Тогда
                            MIMEТип = "image/jpeg";
                        ИначеЕсли Расширение = "png" Тогда
                            MIMEТип = "image/png";
                        ИначеЕсли Расширение = "gif" Тогда
                            MIMEТип = "image/gif";
                        ИначеЕсли Расширение = "webp" Тогда
                            MIMEТип = "image/webp";
                        Иначе
                            MIMEТип = "image/jpeg";
                        КонецЕсли;

                        // Формируем Data URL
                        ИзображениеURL = "data:" + MIMEТип + ";base64," + СтрокаBase64;
                        МассивИзображений.Добавить(ИзображениеURL);
                    КонецЕсли;
                КонецЕсли;
            КонецЦикла;
        КонецЕсли;

    Исключение
        // Если регистр не найден или другая ошибка - пробуем альтернативный способ
    КонецПопытки;

    // Вариант 2: Если изображения хранятся непосредственно в реквизите номенклатуры
    Если МассивИзображений.Количество() = 0 Тогда
        Попытка
            // Проверяем наличие реквизита "Картинка" или "ФайлКартинки"
            Если НоменклатураСсылка.Метаданные().Реквизиты.Найти("Картинка") <> Неопределено Тогда
                Картинка = НоменклатураСсылка.Картинка;
                Если ЗначениеЗаполнено(Картинка) Тогда
                    ДанныеФайла = ПолучитьДвоичныеДанныеФайла(Картинка);
                    Если ДанныеФайла <> Неопределено Тогда
                        СтрокаBase64 = Base64Строка(ДанныеФайла);
                        ИзображениеURL = "data:image/jpeg;base64," + СтрокаBase64;
                        МассивИзображений.Добавить(ИзображениеURL);
                    КонецЕсли;
                КонецЕсли;
            КонецЕсли;
        Исключение
        КонецПопытки;
    КонецЕсли;

    // Вариант 3: Присоединенные файлы (типовой механизм БСП)
    Если МассивИзображений.Количество() = 0 Тогда
        Попытка
            // Для конфигураций на БСП
            МассивФайлов = ПрисоединенныеФайлы.ПолучитьПрисоединенныеФайлы(НоменклатураСсылка);

            Для Каждого ФайлСсылка Из МассивФайлов Цикл
                Расширение = НРег(СокрЛП(ФайлСсылка.Расширение));

                // Проверяем, что это изображение
                Если Расширение = "jpg" ИЛИ Расширение = "jpeg" ИЛИ Расширение = "png"
                    ИЛИ Расширение = "gif" ИЛИ Расширение = "webp" Тогда

                    ДанныеФайла = ПрисоединенныеФайлы.ПолучитьДвоичныеДанныеФайла(ФайлСсылка);

                    Если ДанныеФайла <> Неопределено Тогда
                        СтрокаBase64 = Base64Строка(ДанныеФайла);

                        Если Расширение = "png" Тогда
                            MIMEТип = "image/png";
                        ИначеЕсли Расширение = "gif" Тогда
                            MIMEТип = "image/gif";
                        ИначеЕсли Расширение = "webp" Тогда
                            MIMEТип = "image/webp";
                        Иначе
                            MIMEТип = "image/jpeg";
                        КонецЕсли;

                        ИзображениеURL = "data:" + MIMEТип + ";base64," + СтрокаBase64;
                        МассивИзображений.Добавить(ИзображениеURL);
                    КонецЕсли;
                КонецЕсли;
            КонецЦикла;
        Исключение
        КонецПопытки;
    КонецЕсли;

    Возврат МассивИзображений;

КонецФункции

// =============================================================================
// ПОЛУЧЕНИЕ ДВОИЧНЫХ ДАННЫХ ФАЙЛА
// Вспомогательная функция для получения данных из справочника Файлы
// =============================================================================
Функция ПолучитьДвоичныеДанныеФайла(ФайлСсылка)

    Попытка
        // Для типовых конфигураций с БСП
        Если ОбщегоНазначения.ПодсистемаСуществует("СтандартныеПодсистемы.РаботаСФайлами") Тогда
            МодульРаботаСФайлами = ОбщегоНазначения.ОбщийМодуль("РаботаСФайлами");
            Возврат МодульРаботаСФайлами.ДвоичныеДанныеФайла(ФайлСсылка);
        КонецЕсли;

        // Альтернативный способ - напрямую из хранилища
        Если ТипЗнч(ФайлСсылка) = Тип("СправочникСсылка.Файлы") Тогда
            ФайлОбъект = ФайлСсылка.ПолучитьОбъект();
            Если ФайлОбъект <> Неопределено И ЗначениеЗаполнено(ФайлОбъект.ХранилищеФайла) Тогда
                Возврат ФайлОбъект.ХранилищеФайла.Получить();
            КонецЕсли;
        КонецЕсли;

        // Для присоединенных файлов
        Если СтрНайти(ТипЗнч(ФайлСсылка), "ПрисоединенныеФайлы") > 0 Тогда
            ФайлОбъект = ФайлСсылка.ПолучитьОбъект();
            Если ФайлОбъект <> Неопределено Тогда
                Если ЗначениеЗаполнено(ФайлОбъект.ХранилищеФайла) Тогда
                    Возврат ФайлОбъект.ХранилищеФайла.Получить();
                ИначеЕсли ЗначениеЗаполнено(ФайлОбъект.ПутьКФайлу) Тогда
                    Возврат Новый ДвоичныеДанные(ФайлОбъект.ПутьКФайлу);
                КонецЕсли;
            КонецЕсли;
        КонецЕсли;

    Исключение
    КонецПопытки;

    Возврат Неопределено;

КонецФункции
```
